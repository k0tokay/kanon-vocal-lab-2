cmake_minimum_required(VERSION 3.20)
project(VocalTractLabMac LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets REQUIRED COMPONENTS core base gl)
include(${wxWidgets_USE_FILE})
file(GLOB gui_sources CONFIGURE_DEPENDS GUI/Developer/Sources/Frontend/*.cpp GUI/Developer/Sources/Backend/*.cpp)
add_executable(VocalTractLab2 ${gui_sources})
target_compile_definitions(VocalTractLab2 PRIVATE HAVE_OPENAL GL_SILENCE_DEPRECATION)
target_link_libraries(VocalTractLab2 PRIVATE ${wxWidgets_LIBRARIES} "-framework OpenGL" "-framework OpenAL")
foreach(asset JD2.speaker M01.speaker W02.speaker Vowel-a-outline.gif Vowel-i-outline.gif Vowel-u-outline.gif)
  configure_file(GUI/${asset} ${asset} COPYONLY)
endforeach()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/config.ini")
  configure_file(GUI/config.ini config.ini COPYONLY)
endif()
