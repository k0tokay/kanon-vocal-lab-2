cmake_minimum_required(VERSION 3.20)
project(VocalTractLabMac LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Without a build type CMake compiles with no optimization, which makes the
# synthesis 4-5x slower. Default to Release (also for existing caches that
# were configured with an empty build type).
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
# The GUI needs wxWidgets; without it (e.g. on a Linux server) only the API library is built.
option(VTL_BUILD_GUI "Build the VocalTractLab GUI" ON)
if(VTL_BUILD_GUI)
  find_package(wxWidgets REQUIRED COMPONENTS core base gl)
  include(${wxWidgets_USE_FILE})
endif()
if(APPLE)
  set(VTL_OPENAL "-framework OpenAL")
else()
  find_package(OpenAL REQUIRED)
  set(VTL_OPENAL ${OPENAL_LIBRARY})
endif()
file(GLOB gui_sources CONFIGURE_DEPENDS GUI/Developer/Sources/Frontend/*.cpp)
file(GLOB backend_sources CONFIGURE_DEPENDS backend/*.cpp)
add_library(VocalTractLabApi SHARED ${backend_sources})
target_compile_definitions(VocalTractLabApi PRIVATE HAVE_OPENAL)
target_link_libraries(VocalTractLabApi PRIVATE Threads::Threads ${VTL_OPENAL})
if(NOT APPLE)
  target_include_directories(VocalTractLabApi PRIVATE ${OPENAL_INCLUDE_DIR}/..)
endif()
if(NOT VTL_BUILD_GUI)
  return()
endif()
add_executable(VocalTractLab2 ${gui_sources} ${backend_sources})
target_compile_definitions(VocalTractLab2 PRIVATE HAVE_OPENAL GL_SILENCE_DEPRECATION)
target_link_libraries(VocalTractLab2 PRIVATE Threads::Threads ${wxWidgets_LIBRARIES} "-framework OpenGL" "-framework OpenAL")
foreach(speaker JD2 M01 W02)
  configure_file(speakers/${speaker}.speaker ${speaker}.speaker COPYONLY)
endforeach()
foreach(asset Vowel-a-outline.gif Vowel-i-outline.gif Vowel-u-outline.gif)
  configure_file(GUI/${asset} ${asset} COPYONLY)
endforeach()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/config.ini")
  configure_file(GUI/config.ini config.ini COPYONLY)
endif()
