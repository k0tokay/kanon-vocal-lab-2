cmake_minimum_required(VERSION 3.20)
project(VocalTractLabMac LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets REQUIRED COMPONENTS core base gl)
include(${wxWidgets_USE_FILE})
file(GLOB gui_sources CONFIGURE_DEPENDS GUI/Developer/Sources/Frontend/*.cpp)
file(GLOB backend_sources CONFIGURE_DEPENDS backend/*.cpp)
add_library(VocalTractLabApi SHARED ${backend_sources})
target_compile_definitions(VocalTractLabApi PRIVATE HAVE_OPENAL)
target_link_libraries(VocalTractLabApi PRIVATE "-framework OpenAL")
add_executable(VocalTractLab2 ${gui_sources} ${backend_sources})
target_compile_definitions(VocalTractLab2 PRIVATE HAVE_OPENAL GL_SILENCE_DEPRECATION)
target_link_libraries(VocalTractLab2 PRIVATE ${wxWidgets_LIBRARIES} "-framework OpenGL" "-framework OpenAL")
foreach(speaker JD2 M01 W02)
  configure_file(speakers/${speaker}.speaker ${speaker}.speaker COPYONLY)
endforeach()
foreach(asset Vowel-a-outline.gif Vowel-i-outline.gif Vowel-u-outline.gif)
  configure_file(GUI/${asset} ${asset} COPYONLY)
endforeach()
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/config.ini")
  configure_file(GUI/config.ini config.ini COPYONLY)
endif()
